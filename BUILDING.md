# Building from Source

## Prerequisites

- Steam Deck running SteamOS
- Distrobox with Arch Linux container named "goldeneye"

## Setup Distrobox Container

```bash
distrobox create --name goldeneye --image archlinux:latest
distrobox enter goldeneye -- sudo pacman -S --noconfirm \
  cmake ninja clang pkg-config gtk3 glib2-devel vulkan-headers \
  vulkan-icd-loader alsa-lib python glibc linux-api-headers \
  systemd-libs pipewire pipewire-alsa pulseaudio libpulse git
```

## Clone Repos

```bash
git clone --branch Goldeneye https://github.com/SunJaycy/GoldenEye-Recomp.git
git clone https://github.com/SunJaycy/GoldenEye-Recomp-rexglue.git
```

## Apply Patches

```bash
cd GoldenEye-Recomp-rexglue
git apply ../patches/sdl3_steamdeck_gamepad.patch --directory=thirdparty/sdl3
git apply ../patches/snappy_avx2.patch --directory=thirdparty/snappy
cd ../GoldenEye-Recomp
git apply ../patches/ge_hooks_linux.patch
```

## Build SDK

```bash
distrobox enter goldeneye
cd ~/GoldenEye-Recomp-rexglue
cmake --preset linux-amd64 \
  -DCMAKE_CXX_COMPILER=clang++ -DCMAKE_C_COMPILER=clang \
  -DSDL_JOYSTICK_LINUX=ON -DSDL_LIBUDEV=OFF -DHAVE_LINUX_INPUT_H=ON \
  -DREXGLUE_ENABLE_TRACY=OFF \
  -DCMAKE_CXX_FLAGS="-march=x86-64-v3 $(pkg-config --cflags gtk+-3.0)" \
  -DCMAKE_C_FLAGS="-march=x86-64-v3 $(pkg-config --cflags gtk+-3.0)"

# Apply PCH fix (required after every reconfigure)
printf '/* generated by CMake */\n#pragma clang system_header\n#include <stdlib.h>\n#include <stdio.h>\n#include <errno.h>\n#include <math.h>\n#include <string.h>\n#include <unistd.h>\n#include <pthread.h>\n' \
  > out/build/linux-amd64/thirdparty/sdl3/CMakeFiles/SDL3-static.dir/Release/cmake_pch.h
rm -f out/build/linux-amd64/thirdparty/sdl3/CMakeFiles/SDL3-static.dir/Release/cmake_pch.h.pch

cmake --build --preset linux-amd64-release --parallel
```

## Run Codegen

```bash
cd ~/GoldenEye-Recomp
~/GoldenEye-Recomp-rexglue/out/linux-amd64/Release/rexglue codegen ge_manifest.toml
```

## Build Game

```bash
cmake --preset linux-amd64-release \
  -DCMAKE_CXX_COMPILER=clang++ -DCMAKE_C_COMPILER=clang \
  -DREXSDK_DIR="$HOME/GoldenEye-Recomp-rexglue" \
  -DCMAKE_CXX_FLAGS="-march=x86-64-v3 $(pkg-config --cflags gtk+-3.0)" \
  -DCMAKE_C_FLAGS="-march=x86-64-v3 $(pkg-config --cflags gtk+-3.0)"

# Apply PCH fix to game build dir too
printf '/* generated by CMake */\n#pragma clang system_header\n#include <stdlib.h>\n#include <stdio.h>\n#include <errno.h>\n#include <math.h>\n#include <string.h>\n#include <unistd.h>\n#include <pthread.h>\n' \
  > out/build/linux-amd64-release/rexglue-sdk/thirdparty/sdl3/CMakeFiles/SDL3-static.dir/cmake_pch.h
rm -f out/build/linux-amd64-release/rexglue-sdk/thirdparty/sdl3/CMakeFiles/SDL3-static.dir/cmake_pch.h.pch

cmake --build --preset linux-amd64-release --parallel
```

## Bundle

```bash
mkdir -p goldeneye-bundle
cp out/build/linux-amd64-release/ge goldeneye-bundle/
cp ~/GoldenEye-Recomp-rexglue/out/linux-amd64/Release/librexruntime.so goldeneye-bundle/
cp ~/GoldenEye-Recomp-rexglue/out/linux-amd64/Release/libSPIRV-Tools-shared.so goldeneye-bundle/
cp ~/GoldenEye-Recomp-rexglue/out/linux-amd64/Release/libTracyClient.so goldeneye-bundle/

# Copy glibc from distrobox container
cp /usr/lib/libm.so.6 goldeneye-bundle/
cp /usr/lib/libc.so.6 goldeneye-bundle/
cp /usr/lib/libstdc++.so.6 goldeneye-bundle/
cp /usr/lib/libgcc_s.so.1 goldeneye-bundle/
cp /usr/lib/ld-linux-x86-64.so.2 goldeneye-bundle/
cp /usr/lib/libpipewire-0.3.so* goldeneye-bundle/
cp /usr/lib/libpulse*.so* goldeneye-bundle/
```
